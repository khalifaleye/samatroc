    18→    18→  try {
    19→    19→    if (!existsSync(IMG_DIR)) return [];
    20→    20→    const files = readdirSync(IMG_DIR)
    21→    21→      .filter((f) => f.startsWith(key + "-"))
    22→    22→      .sort()
    23→    23→      .slice(0, max)

    27→    27→    return [];
